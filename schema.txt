-- Enhanced schema additions for TripTo2.0

-- Add tags support for blog posts
create table public.blog_tags (
  tag_id uuid not null default gen_random_uuid(),
  tag_name varchar(50) not null unique,
  color varchar(7) default '#3b82f6',
  created_at timestamp with time zone default now(),
  usage_count integer default 0,
  constraint blog_tags_pkey primary key (tag_id)
);

-- Junction table for blog post tags
create table public.blog_post_tags (
  post_id uuid not null,
  tag_id uuid not null,
  constraint blog_post_tags_pkey primary key (post_id, tag_id),
  constraint blog_post_tags_post_fkey foreign key (post_id) references blogpost (post_id) on delete cascade,
  constraint blog_post_tags_tag_fkey foreign key (tag_id) references blog_tags (tag_id) on delete cascade
);

-- Add reactions support for blog posts
create table public.blog_post_reactions (
  reaction_id uuid not null default gen_random_uuid(),
  post_id uuid not null,
  user_id uuid not null,
  reaction_type varchar(20) not null,
  created_at timestamp with time zone default now(),
  constraint blog_post_reactions_pkey primary key (reaction_id),
  constraint unique_user_post_reaction unique (user_id, post_id, reaction_type),
  constraint blog_post_reactions_post_fkey foreign key (post_id) references blogpost (post_id) on delete cascade,
  constraint blog_post_reactions_user_fkey foreign key (user_id) references user_profiles (user_id) on delete cascade,
  constraint valid_reaction_type check (reaction_type in ('like', 'love', 'laugh', 'wow', 'sad', 'angry'))
);

-- Add hotel support
create table public.hotels (
  hotel_id uuid not null default gen_random_uuid(),
  hotel_name varchar(255) not null,
  location_id uuid not null,
  address text,
  star_rating integer check (star_rating between 1 and 5),
  description text,
  amenities jsonb,
  contact_info jsonb,
  check_in_time time default '15:00:00',
  check_out_time time default '11:00:00',
  created_at timestamp with time zone default now(),
  is_active boolean default true,
  constraint hotels_pkey primary key (hotel_id),
  constraint hotels_location_fkey foreign key (location_id) references locations (location_id)
);

-- Hotel room types
create table public.hotel_rooms (
  room_id uuid not null default gen_random_uuid(),
  hotel_id uuid not null,
  room_type varchar(100) not null,
  bed_type varchar(50),
  capacity integer not null check (capacity > 0),
  base_price numeric(10,2) not null check (base_price >= 0),
  room_size_sqm integer,
  amenities jsonb,
  is_available boolean default true,
  constraint hotel_rooms_pkey primary key (room_id),
  constraint hotel_rooms_hotel_fkey foreign key (hotel_id) references hotels (hotel_id) on delete cascade
);

-- Hotel bookings
create table public.hotel_bookings (
  booking_id uuid not null,
  hotel_id uuid not null,
  room_id uuid not null,
  check_in_date date not null,
  check_out_date date not null,
  guest_count integer not null check (guest_count > 0),
  special_requests text,
  constraint hotel_bookings_pkey primary key (booking_id),
  constraint hotel_bookings_booking_fkey foreign key (booking_id) references booking (booking_id) on delete cascade,
  constraint hotel_bookings_hotel_fkey foreign key (hotel_id) references hotels (hotel_id),
  constraint hotel_bookings_room_fkey foreign key (room_id) references hotel_rooms (room_id),
  constraint valid_date_range check (check_out_date > check_in_date)
);

-- Add user activity tracking
alter table public.user_profiles add column if not exists last_activity timestamp with time zone default now();
alter table public.user_profiles add column if not exists is_online boolean default false;
alter table public.user_profiles add column if not exists role varchar(20) default 'user' check (role in ('user', 'admin', 'moderator'));

-- Add enhanced flight details
alter table public.flight add column if not exists duration_minutes integer;
alter table public.flight add column if not exists has_transit boolean default false;
alter table public.flight add column if not exists transit_airports text[];
alter table public.flight add column if not exists transit_duration_minutes integer[];

-- Add trending support for posts
alter table public.blogpost add column if not exists view_count integer default 0;
alter table public.blogpost add column if not exists is_trending boolean default false;
alter table public.blogpost add column if not exists trending_score numeric default 0;
alter table public.blogpost add column if not exists comment_count integer default 0;

create table public.coupons (
  coupon_id uuid not null default gen_random_uuid (),
  coupon_code character varying(20) not null,
  coupon_type public.coupon_type not null default 'flight_discount'::coupon_type,
  discount_type character varying(10) not null,
  discount_value numeric(10, 2) not null,
  max_discount_amount numeric(10, 2) null,
  min_purchase_amount numeric(10, 2) null default 0,
  usage_limit integer not null default 1,
  usage_count integer not null default 0,
  valid_from timestamp with time zone not null default now(),
  valid_until timestamp with time zone not null,
  created_by uuid null,
  assigned_to_user uuid null,
  package_id uuid null,
  applicable_to_flights boolean not null default true,
  applicable_to_packages boolean not null default false,
  title text not null,
  description text null,
  status public.coupon_status not null default 'active'::coupon_status,
  created_at timestamp with time zone not null default now(),
  updated_at timestamp with time zone not null default now(),
  constraint coupons_pkey primary key (coupon_id),
  constraint coupons_coupon_code_key unique (coupon_code),
  constraint coupons_package_id_fkey foreign KEY (package_id) references travel_package (package_id) on delete CASCADE,
  constraint coupons_created_by_fkey foreign KEY (created_by) references user_profiles (user_id) on delete set null,
  constraint coupons_assigned_to_user_fkey foreign KEY (assigned_to_user) references user_profiles (user_id) on delete CASCADE,
  constraint coupons_valid_period_check check ((valid_until > valid_from)),
  constraint coupons_discount_type_check check (
    (
      (discount_type)::text = any (
        (
          array[
            'percentage'::character varying,
            'fixed'::character varying
          ]
        )::text[]
      )
    )
  ),
  constraint coupons_discount_value_check check ((discount_value > (0)::numeric)),
  constraint coupons_max_discount_check check (
    (
      (max_discount_amount is null)
      or (max_discount_amount > (0)::numeric)
    )
  ),
  constraint coupons_usage_check check ((usage_count <= usage_limit))
) TABLESPACE pg_default;

create index IF not exists idx_coupons_code on public.coupons using btree (coupon_code) TABLESPACE pg_default;

create view public.geography_columns as
select
  current_database() as f_table_catalog,
  n.nspname as f_table_schema,

        'm'::"char",

      ]
    )
  )
  and not pg_is_other_temp_schema(c.relnamespace)
  and has_table_privilege(c.oid, 'SELECT'::text);

create view public.geometry_columns as
select
  current_database()::character varying(256) as f_table_catalog,
  n.nspname as f_table_schema,
  c.relname as f_table_name,
  a.attname as f_geometry_column,
  COALESCE(postgis_typmod_dims (a.atttypmod), sn.ndims, 2) as coord_dimension,
  COALESCE(
    NULLIF(postgis_typmod_srid (a.atttypmod), 0),
    sr.srid,
    0
  ) as srid,
  replace(
    replace(
      COALESCE(
        NULLIF(
          upper(postgis_typmod_type (a.atttypmod)),
          'GEOMETRY'::text
        ),
        st.type,
        'GEOMETRY'::text
      ),
      'ZM'::text,
      ''::text
    ),
    'Z'::text,
    ''::text
  )::character varying(30) as type
from
  pg_class c
  join pg_attribute a on a.attrelid = c.oid
  and not a.attisdropped
  join pg_namespace n on c.relnamespace = n.oid
  join pg_type t on a.atttypid = t.oid
  left join (
    select
      s.connamespace,
      s.conrelid,
      s.conkey,
      replace(
        split_part(s.consrc, ''''::text, 2),
        ')'::text,
        ''::text
      ) as type
    from
      (
        select
          pg_constraint.connamespace,
          pg_constraint.conrelid,
          pg_constraint.conkey,
          pg_get_constraintdef(pg_constraint.oid) as consrc
        from
          pg_constraint
      ) s
    where
      s.consrc ~~* '%geometrytype(% = %'::text
  ) st on st.connamespace = n.oid
  and st.conrelid = c.oid
  and (a.attnum = any (st.conkey))
  left join (
    select
      s.connamespace,
      s.conrelid,
      s.conkey,
      replace(
        split_part(s.consrc, ' = '::text, 2),
        ')'::text,
        ''::text
      )::integer as ndims
    from
      (
        select
          pg_constraint.connamespace,
          pg_constraint.conrelid,
          pg_constraint.conkey,
          pg_get_constraintdef(pg_constraint.oid) as consrc
        from
          pg_constraint
      ) s
    where
      s.consrc ~~* '%ndims(% = %'::text
  ) sn on sn.connamespace = n.oid
  and sn.conrelid = c.oid
  and (a.attnum = any (sn.conkey))
  left join (
    select
      s.connamespace,
      s.conrelid,
      s.conkey,
      replace(
        replace(
          split_part(s.consrc, ' = '::text, 2),
          ')'::text,
          ''::text
        ),
        '('::text,
        ''::text
      )::integer as srid
    from
      (
        select
          pg_constraint.connamespace,
          pg_constraint.conrelid,
          pg_constraint.conkey,
          pg_get_constraintdef(pg_constraint.oid) as consrc
        from
          pg_constraint
      ) s
    where
      s.consrc ~~* '%srid(% = %'::text
  ) sr on sr.connamespace = n.oid
  and sr.conrelid = c.oid
  and (a.attnum = any (sr.conkey))
where
  (
    c.relkind = any (
      array[
        'r'::"char",
        'v'::"char",
        'm'::"char",
        'f'::"char",
        'p'::"char"
      ]
    )
  )
  and not c.relname = 'raster_columns'::name
  and t.typname = 'geometry'::name
  and not pg_is_other_temp_schema(c.relnamespace)
  and has_table_privilege(c.oid, 'SELECT'::text);

BEGIN

        ELSEIF OLD.vote_type = -1 THEN
            UPDATE community_post SET downvote_count = downvote_count - 1 WHERE post_id = OLD.post_id;

            UPDATE community_post SET upvote_count = upvote_count - 1, downvote_count = downvote_count + 1 WHERE post_id = NEW.post_id;
        ELSEIF OLD.vote_type = -1 AND NEW.vote_type = 1 THEN

        END IF;
    END IF;

sync_existing_coupons_on_metadata_insert
DECLARE
    discount_settings JSONB;
    affected_coupons INTEGER := 0;
BEGIN
    -- Only proceed if the flight_discount_settings metadata is being inserted
    IF NEW.metadata_key = 'flight_discount_settings' THEN
        
        -- Extract the discount settings
        discount_settings := NEW.metadata_value;
        
        -- Update any existing unused coupons for this package that were created
        -- before the metadata was set (edge case handling)
        UPDATE coupons 
        SET 
            discount_value = COALESCE((discount_settings->>'discount_percent')::DECIMAL, discount_value),
            max_discount_amount = COALESCE((discount_settings->>'max_discount_amount')::DECIMAL, max_discount_amount),
            valid_until = CASE 
                WHEN (discount_settings->>'validity_days')::INTEGER IS NOT NULL 
                THEN (created_at + INTERVAL '1 day' * (discount_settings->>'validity_days')::INTEGER)
                ELSE valid_until 
            END,
            title = COALESCE(
                (discount_settings->>'discount_percent')::TEXT || '% Off Your Next Flight',
                title
            ),
            description = COALESCE(
                'Enjoy ' || (discount_settings->>'discount_percent')::TEXT || 
                '% discount on flight bookings up to $' || (discount_settings->>'max_discount_amount')::TEXT || 
                ' as a thank you for booking with us!',
                description
            ),
            updated_at = NOW()
        WHERE 
            package_id = NEW.package_id
            AND coupon_type = 'flight_discount'
            AND status = 'active'
            AND valid_until > NOW()
            AND coupon_id NOT IN (
                SELECT DISTINCT coupon_id 
                FROM coupon_usage 
                WHERE coupon_id IS NOT NULL
            );
            
        GET DIAGNOSTICS affected_coupons = ROW_COUNT;
        
        RAISE NOTICE 'Package % discount metadata added. % existing unused coupons synchronized.', 
                     NEW.package_id, affected_coupons;
        
    END IF;
    
    RETURN NEW;
END;


update_unused_coupons_on_package_change
DECLARE
    discount_settings JSONB;
    affected_coupons INTEGER := 0;
BEGIN
    -- Only proceed if the flight_discount_settings metadata is being updated
    IF NEW.metadata_key = 'flight_discount_settings' THEN
        
        -- Extract the new discount settings
        discount_settings := NEW.metadata_value;
        
        -- Update all unused coupons for this package
        -- A coupon is considered "unused" if it hasn't been used (no entry in coupon_usage)
        -- and is still active and not expired
        UPDATE coupons 
        SET 
            discount_value = COALESCE((discount_settings->>'discount_percent')::DECIMAL, discount_value),
            max_discount_amount = COALESCE((discount_settings->>'max_discount_amount')::DECIMAL, max_discount_amount),
            valid_until = CASE 
                WHEN (discount_settings->>'validity_days')::INTEGER IS NOT NULL 
                THEN (created_at + INTERVAL '1 day' * (discount_settings->>'validity_days')::INTEGER)
                ELSE valid_until 
            END,
            title = COALESCE(
                (discount_settings->>'discount_percent')::TEXT || '% Off Your Next Flight',
                title
            ),
            description = COALESCE(
                'Enjoy ' || (discount_settings->>'discount_percent')::TEXT || 
                '% discount on flight bookings up to $' || (discount_settings->>'max_discount_amount')::TEXT || 
                ' as a thank you for booking with us!',
                description
            ),
            updated_at = NOW()
        WHERE 
            package_id = NEW.package_id
            AND coupon_type = 'flight_discount'
            AND status = 'active'
            AND valid_until > NOW()  -- Not expired
            AND coupon_id NOT IN (
                -- Exclude coupons that have already been used
                SELECT DISTINCT coupon_id 
                FROM coupon_usage 
                WHERE coupon_id IS NOT NULL
            );
            
        -- Get count of affected coupons for logging
        GET DIAGNOSTICS affected_coupons = ROW_COUNT;
        
        -- Log the update for audit purposes
        RAISE NOTICE 'Package % discount settings updated. % unused coupons synchronized.', 
                     NEW.package_id, affected_coupons;
        
    END IF;
    
    RETURN NEW;
END;

update_blogpost_vote_counts
BEGIN
    -- This function is called by the trigger on the blog_post_votes table.
    -- It correctly handles all cases: new votes, deleted votes, and changed votes.

    IF (TG_OP = 'INSERT') THEN
        -- A new vote has been cast.
        IF NEW.vote_type = 1 THEN
            UPDATE blogpost SET upvote_count = upvote_count + 1 WHERE post_id = NEW.post_id;
        ELSEIF NEW.vote_type = -1 THEN
            UPDATE blogpost SET downvote_count = downvote_count + 1 WHERE post_id = NEW.post_id;
        END IF;

    ELSIF (TG_OP = 'DELETE') THEN
        -- A vote has been removed (e.g., user un-clicks the button).
        IF OLD.vote_type = 1 THEN
            UPDATE blogpost SET upvote_count = upvote_count - 1 WHERE post_id = OLD.post_id;
        ELSEIF OLD.vote_type = -1 THEN
            UPDATE blogpost SET downvote_count = downvote_count - 1 WHERE post_id = OLD.post_id;
        END IF;

    ELSIF (TG_OP = 'UPDATE') THEN
        -- A user has changed their vote (e.g., from upvote to downvote).
        IF OLD.vote_type = 1 AND NEW.vote_type = -1 THEN
            -- Changed from UPVOTE to DOWNVOTE: Decrement upvotes, increment downvotes.
            UPDATE blogpost SET upvote_count = upvote_count - 1, downvote_count = downvote_count + 1 WHERE post_id = NEW.post_id;
        ELSEIF OLD.vote_type = -1 AND NEW.vote_type = 1 THEN
            -- Changed from DOWNVOTE to UPVOTE: Decrement downvotes, increment upvotes.
            UPDATE blogpost SET downvote_count = downvote_count - 1, upvote_count = upvote_count + 1 WHERE post_id = NEW.post_id;
        END IF;
    END IF;

    RETURN NULL; -- The result is ignored since this is an AFTER trigger.
END;

update_coupon_usage_count
BEGIN
    -- Increment usage count
    UPDATE coupons 
    SET usage_count = usage_count + 1,
        updated_at = NOW()
    WHERE coupon_id = NEW.coupon_id;
    
    -- Mark as used if reached usage limit
    UPDATE coupons 
    SET status = 'used',
        updated_at = NOW()
    WHERE coupon_id = NEW.coupon_id 
    AND usage_count >= usage_limit;
    
    RETURN NEW;
END;
