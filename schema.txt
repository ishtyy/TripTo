-- This table will track each user's vote on a community post.
CREATE TABLE public.community_post_votes (
    vote_id uuid NOT NULL DEFAULT gen_random_uuid(),
    post_id uuid NOT NULL,
    user_id uuid NOT NULL,
    vote_type smallint NOT NULL, -- 1 for upvote, -1 for downvote
    CONSTRAINT community_post_votes_pkey PRIMARY KEY (vote_id),
    CONSTRAINT community_post_votes_post_id_fkey FOREIGN KEY (post_id) REFERENCES public.community_post(post_id) ON DELETE CASCADE,
    CONSTRAINT community_post_votes_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.user_profiles(user_id) ON DELETE CASCADE,
    -- A user can only vote once on a post.
    CONSTRAINT unique_user_post_vote UNIQUE (user_id, post_id),
    CONSTRAINT vote_type_check CHECK ((vote_type = 1) OR (vote_type = -1))
);
